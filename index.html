void main() { Scanner teclado = new Scanner(System.in); Random random = new Random(); int rng; int rngplayer; int saldo; int aposta; int escolha; saldo = 100; do { System.out.println("Bem vindo ao cassino!\nEscolha uma das opções a seguir:\n1-Apostar\n2-Ver Saldo\n3-Sair"); escolha = teclado.nextInt(); rng = random.nextInt(0,3); switch (escolha) { case 1: System.out.println("Você quer apostar quantos reais no cassino(odds=2x)?"); aposta = teclado.nextInt(); if (aposta > saldo) { System.out.println("Aposta inválida"); } else { System.out.println("Escolha em quem apostar: 0 para brancas, 1 para pretas e 2 para vermelhas"); rngplayer = teclado.nextInt(); if (rngplayer == rng) { System.out.println("Você acertou!"); saldo = saldo -aposta+(aposta * 2); } else { System.out.println("Você errou"); saldo = saldo - aposta; } } break; case 2: System.out.println("O seu saldo é:" + saldo); break; case 3: System.out.println("Até mais"); break; default: System.out.println("Escolha inválida"); break; } } while (escolha != 3); }
